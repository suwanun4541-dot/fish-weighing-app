// --------- Code.gs --------- const REPORT_SHEET_NAME = "รายงาน"; const MORTALITY_SHEET_NAME = "ปลาตาย"; const KV_SHEET_NAME = "Records"; function getOrCreateSheet(name, headerRow) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); if (headerRow) sheet.appendRow(headerRow); } return sheet; } function getKvSheet() { return getOrCreateSheet(KV_SHEET_NAME, ["key", "value", "updatedAt"]); } function kvGet(key) { const sheet = getKvSheet(); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === key) return data[i][1]; } return null; } function kvList(prefix) { const sheet = getKvSheet(); const data = sheet.getDataRange().getValues(); const items = {}; for (let i = 1; i < data.length; i++) { const k = String(data[i][0]); if (!prefix || k.indexOf(prefix) === 0) items[k] = data[i][1]; } return items; } function kvSet(key, value) { const sheet = getKvSheet(); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === key) { sheet.getRange(i + 1, 2).setValue(value); sheet.getRange(i + 1, 3).setValue(new Date()); return; } } sheet.appendRow([key, value, new Date()]); } function kvDelete(key) { const sheet = getKvSheet(); const data = sheet.getDataRange().getValues(); for (let i = data.length - 1; i >= 1; i--) { if (String(data[i][0]) === key) { sheet.deleteRow(i + 1); return; } } } function appendReportRows(rows) { const sheet = getOrCreateSheet(REPORT_SHEET_NAME, [ "วันที่", "เวลา", "สาขา", "บ่อ/กระชัง", "รุ่น", "ขนส่ง", "ทะเบียน", "จำนวนถัง", "นน.รวม", "หักน้ำ", "นน.สุทธิ", "ไซส์(กรัม)", "จำนวนตัว", "หมายเหตุ" ]); rows.forEach(function (r) { sheet.appendRow([ r.dateStr, r.timeStr, r.branch, r.pond, r.batch, r.transport, r.plate, r.bucketCount, r.grossKg, r.waterDeducted, r.netKg, r.sizeGram, r.fishCount, r.note ]); }); } function appendMortalityRows(rows) { const sheet = getOrCreateSheet(MORTALITY_SHEET_NAME, [ "วันที่", "เวลา", "บ่อ/กระชัง", "รุ่น", "จำนวนปลาตาย(ตัว)", "หมายเหตุ" ]); rows.forEach(function (r) { sheet.appendRow([ r.dateStr, r.timeStr, r.pond, r.batch, r.deadCount, r.note ]); }); } function doGet(e) { try { const action = e.parameter.action; if (action === "get") { const value = kvGet(e.parameter.key); return jsonOutput({ value: value === null ? null : value }); } if (action === "list") { const items = kvList(e.parameter.prefix || ""); return jsonOutput({ items: items }); } return jsonOutput({ error: "unknown action" }); } catch (err) { return jsonOutput({ error: String(err) }); } } function doPost(e) { try { const body = JSON.parse(e.postData.contents); if (body.rows) { if (body.type === "mortality") { appendMortalityRows(body.rows); } else { appendReportRows(body.rows); } } if (body.action === "set") { kvSet(body.key, body.value); } else if (body.action === "delete") { kvDelete(body.key); } return jsonOutput({ ok: true }); } catch (err) { return jsonOutput({ ok: false, error: String(err) }); } } function jsonOutput(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
